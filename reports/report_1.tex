\chapter{GIỚI THIỆU ĐỀ TÀI VÀ BÀI TOÁN}

\section{Tên đề tài}

\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Tiếng Việt:} \textit{EcoSort - Ứng dụng di động phân loại rác thông minh hỗ trợ phân loại rác tại nguồn cho hộ gia đình.}
    \item \textbf{Tiếng Anh:} \textit{EcoSort - An Intelligent Waste Classification Mobile Application for Household Source Separation.}
\end{itemize}

\section{Bối cảnh và vấn đề cần giải quyết}

Luật Bảo vệ môi trường 2020 (có hiệu lực đầy đủ từ 01/01/2025) quy định bắt buộc phân loại rác thải sinh hoạt tại nguồn thành ba nhóm chính: \textit{(i)} rác có khả năng tái chế, \textit{(ii)} rác thực phẩm, và \textit{(iii)} rác thải sinh hoạt khác. Tuy nhiên, phần lớn hộ gia đình tại các đô thị lớn (Hà Nội, TP.HCM, Đà Nẵng) vẫn gặp khó khăn do:

\begin{itemize}[leftmargin=1.5cm]
    \item Không nhớ hoặc không chắc chắn vật phẩm cụ thể (vỏ hộp sữa, túi ni-lông, pin, khẩu trang y tế,\ldots) thuộc nhóm nào.
    \item Hướng dẫn hiện có chủ yếu là văn bản/áp phích tĩnh, khó tra cứu tức thời tại thùng rác.
    \item Thiếu phản hồi/động viên nên hành vi phân loại không duy trì được lâu.
\end{itemize}

\section{Ý nghĩa thực tiễn}

Giúp người dân phân loại rác đúng chỉ trong vài giây bằng camera điện thoại; giảm tỷ lệ phân loại sai, hạ chi phí tái phân loại cho đơn vị thu gom; đồng thời nâng cao nhận thức môi trường - phù hợp với lộ trình xử phạt vi phạm phân loại rác đang được triển khai tại các địa phương.

\chapter{NGƯỜI DÙNG}

\section{Nhóm người dùng mục tiêu}

\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Chính (Primary):} Người nội trợ và nhân viên văn phòng 22-55 tuổi tại đô thị, sử dụng smartphone Android/iOS, có nhu cầu phân loại rác đúng để tránh bị nhắc nhở/phạt.
    \item \textbf{Phụ (Secondary):} Học sinh tiểu học - THCS (học qua trò chơi); người cao tuổi (ưu tiên giọng nói, chữ to).
\end{itemize}

\section{Persona tiêu biểu}

\textbf{Persona 1 - Chị Lan, 34 tuổi, nhân viên marketing, mẹ 2 con.} Bận rộn, muốn phân loại nhanh nhưng không nhớ hộp sữa Milo có cần rửa sạch rồi mới cho vào rác tái chế không. Mong muốn: \textit{``chỉ cần chụp 1 phát biết ngay''.}

\textbf{Persona 2 - Bác Tâm, 62 tuổi, hưu trí.} Mắt kém, ngại gõ chữ. Mong muốn: \textit{``ứng dụng tự đọc to kết quả, chữ to.''}

\section{Nhu cầu và kịch bản sử dụng}

\begin{enumerate}[label=\textbf{S\arabic*.}, leftmargin=1.5cm]
    \item \textbf{Phân loại nhanh:} Người dùng mở app $\rightarrow$ chụp hoặc đưa camera vào vật $\rightarrow$ app trả về nhóm rác + mức tin cậy + hướng dẫn xử lý (vd. ``\textit{Rác tái chế - Hãy rửa sạch và bóp dẹt''}).
    \item \textbf{Tra cứu thủ công:} Người dùng gõ/nói tên vật phẩm $\rightarrow$ app trả về nhóm + ví dụ ảnh.
    \item \textbf{Học và chơi:} Mini-game ``phân loại rác'' 60 giây, bảng xếp hạng gia đình, tích điểm đổi huy hiệu nhằm duy trì hành vi.
    \item \textbf{Phản hồi và học tiếp:} Nếu kết quả sai, người dùng chọn nhãn đúng $\rightarrow$ hệ thống ghi nhận để cải thiện (active learning).
\end{enumerate}

\chapter{GIẢI PHÁP}

\section{Ý tưởng hệ thống}

EcoSort là ứng dụng \textbf{di động (Android, có thể mở rộng iOS)} kết hợp \textbf{thị giác máy tính} và \textbf{tương tác đa phương thức (camera + giọng nói + chạm)}. Người dùng chỉ cần hướng camera vào vật phẩm, hệ thống nhận dạng và trả về nhóm rác theo quy định Việt Nam, kèm hướng dẫn xử lý ngắn, phát âm thanh (TTS) và hình ảnh minh họa.

\section{Chức năng chính (MVP ở giai đoạn 1)}

\begin{enumerate}[label=\textbf{F\arabic*.}, leftmargin=1.5cm]
    \item \textbf{Nhận dạng ảnh thời gian thực:} CNN pretrained chạy on-device (TensorFlow Lite) phân loại 3 nhóm chính + 15-20 lớp con phổ biến ở Việt Nam.
    \item \textbf{Hướng dẫn xử lý:} Bảng tri thức (rule-based knowledge base JSON) ánh xạ mỗi lớp con sang nhóm rác và lời khuyên.
    \item \textbf{Tương tác đa phương thức:} Nhập bằng ảnh/giọng nói (Vietnamese ASR), xuất bằng chữ + TTS, hỗ trợ chế độ chữ to - tương phản cao.
    \item \textbf{Phản hồi/sửa lỗi:} Nút ``Kết quả này sai'' $\rightarrow$ người dùng chọn nhãn đúng; dữ liệu được lưu để tinh chỉnh mô hình (có xin phép).
    \item \textbf{Gamification cơ bản:} Điểm, chuỗi ngày (streak), huy hiệu khi phân loại đúng.
\end{enumerate}

\section{Luồng tương tác dự kiến}

\textbf{Luồng chính:} \textit{Mở app $\rightarrow$ Camera $\rightarrow$ Chụp/giữ $\rightarrow$ Suy luận on-device ($<$1 s) $\rightarrow$ Hiển thị thẻ kết quả (nhóm rác, màu đại diện, hướng dẫn, nút phát âm) $\rightarrow$ Người dùng xác nhận / báo sai $\rightarrow$ Cộng điểm.}

Thiết kế theo nguyên lý HCI của Nielsen: phản hồi tức thì ($<$1 s), nhìn thấy trạng thái hệ thống (thanh tin cậy \%), khả năng khôi phục lỗi (nút báo sai), nhất quán màu sắc với quy ước phân loại rác (xanh lá - hữu cơ, cam - tái chế, xám - khác).

\chapter{ĐIỂM NỔI BẬT}

\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Mô hình thị giác máy fine-tune cho ngữ cảnh Việt Nam:} Thu thập và gán nhãn bộ dữ liệu rác đặc thù (hộp xôi xốp, lá chuối, vỏ bánh chưng, chai nước mắm,\ldots) - chưa có trong các dataset quốc tế (TrashNet, TACO).
    \item \textbf{Hybrid AI + Knowledge Base:} CNN nhận dạng vật thể; knowledge base rule-based ánh xạ sang \textit{nhóm rác theo quy định Việt Nam} + hướng dẫn xử lý. Tách rời giúp cập nhật quy định mới mà không cần retrain mô hình.
    \item \textbf{Tương tác đa phương thức có chủ đích:} Camera là kênh chính; giọng nói (ASR + TTS) là kênh phụ cho người cao tuổi/khi tay bẩn. Có cơ chế lựa chọn phương thức thích ứng theo cảm biến môi trường (độ sáng, tiếng ồn).
    \item \textbf{Human-in-the-loop learning:} Nút ``báo sai'' thu thập dữ liệu được người dùng xác nhận, dùng cho vòng fine-tune tiếp theo - phù hợp định hướng \textit{Human-AI Interaction}.
    \item \textbf{Explainability nhẹ:} Hiển thị bản đồ nhiệt Grad-CAM (``\textit{model đang nhìn vào vùng nào}'') như một tùy chọn ở chế độ ``Giải thích'' - tăng độ tin cậy.
    \item \textbf{Gamification + nhật ký cá nhân:} Điểm, chuỗi ngày, huy hiệu, biểu đồ ``\textit{bạn đã tái chế bao nhiêu vật tuần này}'' - khuyến khích duy trì hành vi (hướng \textit{Digital Wellbeing/Behavior Change}).
\end{itemize}

\chapter{DỮ LIỆU}

\section{Dữ liệu công khai - khởi tạo nhanh}

\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{TrashNet} (Stanford): 2.527 ảnh, 6 lớp (glass, paper, cardboard, plastic, metal, trash).
    \item \textbf{TACO} (Trash Annotations in Context): 1.500+ ảnh ngoài trời, có nhãn phân đoạn.
    \item \textbf{Garbage Classification (Kaggle)}: $\sim$15.000 ảnh, 12 lớp.
\end{itemize}

Mục đích: pretrain/benchmark baseline, kiểm tra pipeline sớm.

\section{Dữ liệu tự xây - tạo giá trị khác biệt}

\textbf{EcoSort-VN mini dataset (mục tiêu $\geq$ 3.000 ảnh, 20 lớp)}:

\begin{itemize}[leftmargin=1.5cm]
    \item Mỗi thành viên chụp 150-200 ảnh/tuần tại nhà, chợ, văn phòng (nhiều góc, nhiều ánh sáng).
    \item Tăng cường dữ liệu (data augmentation): xoay, lật, thay nền, nhiễu màu, cutout để nhân rộng x5-x10.
    \item Gán nhãn bằng công cụ Label Studio (miễn phí) theo quy ước nhóm.
    \item Chia 70/15/15 cho train/val/test; cân bằng số lượng mẫu giữa các lớp.
\end{itemize}

\section{Dữ liệu người dùng (sau MVP)}

Thu thập ảnh và nhãn do người dùng xác nhận (nút ``báo sai'') theo cơ chế \textit{opt-in} có xin phép rõ ràng, lưu ẩn danh, dùng để tinh chỉnh mô hình. Tuân thủ nguyên tắc đạo đức: không chụp mặt người, có thể từ chối bất kỳ lúc nào.

\section{Đánh giá người dùng (Usability, 8-10 người)}

Mỗi người tham gia thực hiện 4 tác vụ (chụp phân loại chai nhựa, lá chuối, vỏ pin, túi ni-lông bẩn), sau đó điền phiếu.

\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Task completion rate} (tỷ lệ hoàn thành đúng).
    \item \textbf{Task time} (thời gian trung bình/lần phân loại). Mục tiêu $\leq$ 8 giây.
    \item \textbf{Error rate} và số lần cần giúp đỡ.
    \item \textbf{SUS - System Usability Scale} (10 câu). Mục tiêu $\geq$ 72 (``good'').
    \item \textbf{Phỏng vấn ngắn} (3-5 phút) để lấy phản hồi định tính.
\end{itemize}

\section{Đánh giá tương tác đa phương thức}

Tỷ lệ nhận dạng giọng nói đúng (ASR WER), thời gian phản hồi, tỷ lệ chuyển kênh (voice $\rightarrow$ touch) khi gặp lỗi.
